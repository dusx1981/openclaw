### 3.2 Webhook 模式

```typescript
export async function monitorWebhook(params: MonitorTransportParams): Promise<void> {
  const server = http.createServer();

  server.on("request", (req, res) => {
    // 1. 限流检查
    if (!applyBasicWebhookRequestGuards(...)) return;

    // 2. 读取请求体
    const rawBody = await readRequestBodyWithLimit(req, ...);

    // 3. 签名验证
    if (!isFeishuWebhookSignatureValid({ headers, rawBody, encryptKey })) {
      respondText(res, 401, "Invalid signature");
      return;
    }

    // 4. 处理挑战请求 (URL 验证)
    const { isChallenge, challenge } = Lark.generateChallenge(payload, ...);
    if (isChallenge) {
      res.end(JSON.stringify(challenge));
      return;
    }

    // 5. 分发事件
    const value = await eventDispatcher.invoke(
      buildFeishuWebhookEnvelope(req, payload),
      { needCheck: false }
    );
    res.end(JSON.stringify(value));
  });

  server.listen(port, host);
}
```

**签名验证逻辑**:

```typescript
function isFeishuWebhookSignatureValid(params): boolean {
  const { headers, rawBody, encryptKey } = params;

  const timestamp = headers["x-lark-request-timestamp"];
  const nonce = headers["x-lark-request-nonce"];
  const signature = headers["x-lark-signature"];

  // 计算签名: SHA256(timestamp + nonce + encryptKey + rawBody)
  const computedSignature = crypto
    .createHash("sha256")
    .update(timestamp + nonce + encryptKey + rawBody)
    .digest("hex");

  return safeEqualSecret(computedSignature, signature);
}
```

### 4.3 消息防抖机制

```typescript
const inboundDebouncer = core.channel.debounce.createInboundDebouncer({
  debounceMs: inboundDebounceMs, // 默认 500ms
  buildKey: (event) => {
    // 按会话+发送者构建去重键
    const chatId = event.message.chat_id;
    const senderId = event.sender.sender_id.open_id;
    const rootId = event.message.root_id;
    const threadKey = rootId ? `thread:${rootId}` : "chat";
    return `feishu:${accountId}:${chatId}:${threadKey}:${senderId}`;
  },
  shouldDebounce: (event) => {
    // 仅文本消息且不含控制命令时防抖
    if (event.message.message_type !== "text") return false;
    const text = resolveDebounceText(event);
    return !core.channel.text.hasControlCommand(text, cfg);
  },
  onFlush: async (entries) => {
    // 合并多条消息为一条
    const combinedText = entries.map((e) => resolveDebounceText(e)).join("\n");
    await dispatchFeishuMessage({ ...lastEntry, content: combinedText });
  },
});
```

```typescript
export function parseFeishuMessageEvent(
  event: FeishuMessageEvent,
  botOpenId?: string,
): FeishuMessageContext {
  // 解析消息内容
  const rawContent = parseMessageContent(event.message.content, event.message.message_type);

  // 检查 Bot 是否被提及
  const mentionedBot = checkBotMentioned(event, botOpenId);

  // 规范化提及标签
  const content = normalizeMentions(rawContent, event.message.mentions, botOpenId);

  return {
    chatId: event.message.chat_id,
    messageId: event.message.message_id,
    senderId: event.sender.sender_id.open_id,
    senderOpenId: event.sender.sender_id.open_id,
    chatType: event.message.chat_type, // "p2p" | "group" | "private"
    mentionedBot,
    content,
    rootId: event.message.root_id,
    parentId: event.message.parent_id,
    threadId: event.message.thread_id,
  };
}
```

### 5.2 消息处理主流程

```typescript
export async function handleFeishuMessage(params) {
  const { cfg, event, botOpenId, accountId } = params;

  // 1. 解析消息上下文
  let ctx = parseFeishuMessageEvent(event, botOpenId);
  const isGroup = ctx.chatType === "group";

  // 2. 去重最终确认
  if (!(await finalizeFeishuMessageProcessing(messageId, accountId, log))) {
    return;
  }

  // 3. 解析发送者名称 (可选)
  if (feishuCfg?.resolveSenderNames ?? true) {
    const { name, permissionError } = await resolveFeishuSenderName({
      account, senderId: ctx.senderOpenId, log
    });
    if (name) ctx = { ...ctx, senderName: name };
    if (permissionError) permissionErrorForAgent = permissionError;
  }

  // 4. 群组权限检查
  if (isGroup) {
    // 4.1 检查群组是否在白名单
    const groupPolicy = resolveFeishuGroupPolicy(...);
    const groupAllowed = isFeishuGroupAllowed({ groupPolicy, allowFrom, senderId: ctx.chatId });
    if (!groupAllowed) return;

    // 4.2 检查发送者是否在白名单
    const senderAllowed = isFeishuGroupAllowed({ groupPolicy: "allowlist", allowFrom, senderId: ctx.senderOpenId });
    if (!senderAllowed) return;

    // 4.3 检查是否需要提及 Bot
    const { requireMention } = resolveFeishuReplyPolicy(...);
    if (requireMention && !ctx.mentionedBot) {
      // 记录到历史但不处理
      recordPendingHistoryEntryIfEnabled(...);
      return;
    }
  }

  // 5. 私信权限检查 (配对机制)
  if (!isGroup && dmPolicy === "pairing") {
    const pairing = createChannelPairingController({ core, channel: "feishu", accountId });
    await pairing.issueChallenge({
      senderId: ctx.senderOpenId,
      sendPairingReply: async (text) => {
        await sendMessageFeishu({ cfg, to: `chat:${ctx.chatId}`, text, accountId });
      },
    });
  }

  // 6. 解析群组会话作用域
  const groupSession = isGroup ? resolveFeishuGroupSession({
    chatId: ctx.chatId,
    senderOpenId: ctx.senderOpenId,
    groupConfig,
    feishuCfg,
  }) : null;

  // 7. 解析 Agent 路由
  let route = core.channel.routing.resolveAgentRoute({
    cfg,
    channel: "feishu",
    accountId: account.accountId,
    peer: { kind: isGroup ? "group" : "direct", id: peerId },
    parentPeer,
  });

  // 8. 动态 Agent 创建 (私信场景)
  if (!isGroup && route.matchedBy === "default") {
    const result = await maybeCreateDynamicAgent({ cfg, senderOpenId, dynamicCfg });
    if (result.created) {
      effectiveCfg = result.updatedCfg;
      route = core.channel.routing.resolveAgentRoute({ cfg: effectiveCfg, ... });
    }
  }

  // 9. 构建消息体
  const messageBody = buildFeishuAgentBody({ ctx, quotedContent, permissionErrorForAgent, botOpenId });

  // 10. 解析媒体附件
  const mediaList = await resolveFeishuMediaList({
    cfg, messageId, messageType, content, maxBytes, accountId
  });
  const mediaPayload = buildAgentMediaPayload(mediaList);

  // 11. 分发到 Agent
  if (broadcastAgents) {
    // 广播模式: 分发到多个 Agent
    for (const agentId of broadcastAgents) {
      await dispatchForAgent(agentId);
    }
  } else {
    // 单 Agent 模式
    const ctxPayload = await buildCtxPayloadForAgent(route.agentId, route.sessionKey, ...);
    const { dispatcher, replyOptions } = createFeishuReplyDispatcher({
      cfg, agentId: route.agentId, chatId, replyToMessageId, ...
    });
    await core.channel.reply.dispatchReplyFromConfig({ ctx: ctxPayload, cfg, dispatcher, replyOptions });
  }
}
```

## 6. 回复分发机制

### 6.1 回复分发器创建

**文件**: `extensions/feishu/src/reply-dispatcher.ts`

```typescript
export function createFeishuReplyDispatcher(params: CreateFeishuReplyDispatcherParams) {
  const { cfg, agentId, chatId, replyToMessageId, mentionTargets, accountId, identity } = params;

  // 打字指示器回调
  const { typingCallbacks } = createChannelReplyPipeline({
    typing: {
      start: async () => {
        typingState = await addTypingIndicator({ cfg, messageId: replyToMessageId, accountId });
      },
      stop: async () => {
        await removeTypingIndicator({ cfg, state: typingState, accountId });
        typingState = null;
      },
    },
  });

  // 创建回复分发器
  const { dispatcher, replyOptions } = core.channel.reply.createReplyDispatcherWithTyping({
    onReplyStart: async () => {
      if (streamingEnabled) startStreaming();
      await typingCallbacks?.onReplyStart?.();
    },

    deliver: async (payload: ReplyPayload, info) => {
      const { text, hasMedia } = resolveSendableOutboundReplyParts(payload);

      // 流式卡片处理
      if (streaming?.isActive()) {
        if (info?.kind === "final") {
          await closeStreaming();
        }
        return;
      }

      // 卡片/纯文本选择
      const useCard = renderMode === "card" || (renderMode === "auto" && shouldUseCard(text));

      if (useCard) {
        await sendStructuredCardFeishu({
          cfg,
          to: chatId,
          text,
          replyToMessageId,
          replyInThread,
          mentions: mentionTargets,
          header: resolveCardHeader(agentId, identity),
          note: resolveCardNote(agentId, identity, prefixContext),
        });
      } else {
        await sendMessageFeishu({
          cfg,
          to: chatId,
          text,
          replyToMessageId,
          replyInThread,
          mentions: mentionTargets,
        });
      }

      if (hasMedia) {
        await sendMediaReplies(payload);
      }
    },

    onIdle: async () => {
      await closeStreaming();
      typingCallbacks?.onIdle?.();
    },
  });

  return { dispatcher, replyOptions };
}
```

```typescript
class FeishuStreamingSession {
  async start(chatId, receiveIdType, options) {
    // 创建初始卡片
    const card = buildStreamingCard({ header, note, initialContent: "" });
    const response = await client.im.message.create({
      params: { receive_id_type: receiveIdType },
      data: { receive_id: chatId, content: JSON.stringify(card), msg_type: "interactive" },
    });
    this.messageId = response.data.message_id;
  }

  async update(content: string) {
    // 更新卡片内容
    const card = buildStreamingCard({ content });
    await client.im.message.patch({
      path: { message_id: this.messageId },
      data: { content: JSON.stringify(card) },
    });
  }

  async close(finalContent: string, options) {
    // 最终更新并锁定卡片
    await this.update(finalContent);
  }
}
```

```typescript
export async function sendMessageFeishu(params: {
  cfg: ClawdbotConfig;
  to: string; // "user:<openId>" 或 "chat:<chatId>"
  text: string;
  accountId?: string;
  replyToMessageId?: string;
  replyInThread?: boolean;
  mentions?: MentionTarget[];
}): Promise<FeishuSendResult> {
  const account = resolveFeishuAccount({ cfg, accountId });
  const client = createFeishuClient(account);

  // 解析目标
  const { receiveId, receiveIdType } = resolveFeishuSendTarget(params.to);

  // 构建消息内容
  const content = params.mentions
    ? JSON.stringify(buildMentionedMessage(params.mentions, params.text))
    : JSON.stringify({ text: params.text });

  // 发送回复或直接消息
  if (params.replyToMessageId) {
    return sendReplyOrFallbackDirect(client, {
      replyToMessageId: params.replyToMessageId,
      replyInThread: params.replyInThread,
      content,
      msgType: "text",
      directParams: { receiveId, receiveIdType, content, msgType: "text" },
    });
  }

  const response = await client.im.message.create({
    params: { receive_id_type: receiveIdType },
    data: { receive_id: receiveId, content, msg_type: "text" },
  });

  return toFeishuSendResult(response, receiveId);
}
```

```typescript
export async function sendStructuredCardFeishu(params: {

  to;
  text;

  replyInThread?;

  note?;

      // 内容 (Markdown)

      // 底部备注
      ...(params.note
        ? [{ tag: "note", elements: [{ tag: "plain_text", content: params.note }] }]
        : []),
    ],
  };

  return sendMessageWithContent({ ...params, content, msgType: "interactive" });
```

```typescript
export async function sendMediaFeishu(params: {

  to;
```

### 11.3 常见问题

**问题**: 消息未触发 Agent 响应

- 检查 `allowFrom` / `groupAllowFrom` 配置
- 群组检查 `groupPolicy` 和是否提及 Bot
- 私信检查 `dmPolicy` 配置和配对状态

**问题**: Webhook 签名验证失败

- 确认 `encryptKey` 配置正确
- 检查时间戳是否在有效期内

**问题**: 消息重复处理

- 检查 `dedup` 目录下是否有残留文件
- 重启 Gateway 清理内存去重状态
