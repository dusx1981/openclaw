# 飞书消息转发到微信技术方案

## 1. 背景与需求

将飞书（Feishu）接收到的消息，自动转发到微信（WeChat）。

**约束条件**：

- 微信插件 (`@tencent-weixin/openclaw-weixin`) 仅支持**私聊**（private chats only）
- 微信通过 **QR 扫码登录**认证
- 飞书支持 WebSocket 和 Webhook 两种模式

---

## 2. 方案对比

### 方案概览

| 方案       | 实现方式             | 工作量  | 适用场景            |
| ---------- | -------------------- | ------- | ------------------- |
| **方案 A** | Internal Hook        | 1-2 天  | 简单转发，无状态    |
| **方案 B** | Agent + Message 工具 | 2-3 天  | 需要处理/转换消息   |
| **方案 C** | Webhook + 外部服务   | 3-5 天  | 需要外部系统集成    |
| **方案 D** | 插件开发             | 5-10 天 | 需要复杂逻辑/商业化 |

#### 步骤 3：创建 handler.ts

```typescript
// ~/.openclaw/hooks/feishu-to-wechat/handler.ts

const handler = async (event) => {
  // 1. 只处理飞书消息
  if (event.type !== "message" || event.action !== "received") {
    return;
  }

  const { context } = event;

  // 只转发飞书通道的消息
  if (context.channelId !== "feishu") {
    return;
  }

  // 2. 获取消息内容
  const { from, content, senderName, conversationId } = context;

  // 可选：过滤特定群组或用户
  // if (conversationId !== "oc_allowed_group") return;

  // 3. 构建转发消息
  const forwardMessage = `【飞书转发】${senderName || from}:\n${content}`;

  // 4. 调用 Gateway API 发送到微信
  const response = await fetch("http://localhost:18789/api/message", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      Authorization: `Bearer ${process.env.OPENCLAW_HOOK_TOKEN}`,
    },
    body: JSON.stringify({
      channel: "wechat", // 或 "openclaw-weixin"
      target: process.env.WECHAT_TARGET_USER, // 微信用户 ID
      message: forwardMessage,
    }),
  });

  if (!response.ok) {
    console.error(`[feishu-to-wechat] 发送失败:`, await response.text());
  }
};

export default handler;
```

```typescript
// relay-service.ts (独立服务)
import express from "express";

const app = express();

app.post("/forward", async (req, res) => {
  const { type, action, context } = req.body;

  // 验证 token
  if (req.headers["x-openclaw-token"] !== process.env.WEBHOOK_TOKEN) {
    return res.status(401).send("Unauthorized");
  }

  // 只处理飞书消息
  if (context.channelId !== "feishu") {
    return res.status(200).send("Ignored");
  }

  // 转发到微信
  await sendToWeChat({
    to: process.env.WECHAT_TARGET,
    message: `【飞书】${context.senderName}: ${context.content}`,
  });

  res.status(200).send("OK");
});

app.listen(3000);
```

````
#### handler.ts

```typescript
// ~/.openclaw/hooks/feishu-to-wechat/handler.ts

const GATEWAY_URL = process.env.GATEWAY_URL || "http://localhost:18789";

interface MessageContext {
  from: string;
  content: string;
  channelId: string;
  senderName?: string;
  conversationId?: string;
  isGroup?: boolean;
}

async function forwardToWeChat(context: MessageContext): Promise<void> {
  const targetUser = process.env.WECHAT_TARGET_USER;
  if (!targetUser) {
    console.error("[feishu-to-wechat] 未配置 WECHAT_TARGET_USER");
    return;
  }

  // 构建转发消息
  const prefix = process.env.FORWARD_PREFIX || "【飞书转发】";
  const sender = context.senderName || context.from;
  const location = context.isGroup ? `[群:${context.conversationId}]` : "[私信]";

  const forwardMessage = [
    prefix,
    `${location} ${sender}:`,
    context.content,
  ].join("\n");

  // 调用 Gateway API
  try {
    const response = await fetch(`${GATEWAY_URL}/api/message`, {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "Authorization": `Bearer ${process.env.OPENCLAW_HOOK_TOKEN}`,
      },
      body: JSON.stringify({
        channel: "wechat",
        target: targetUser,
        message: forwardMessage,
      }),
    });

    if (!response.ok) {
      const error = await response.text();
      console.error(`[feishu-to-wechat] 发送失败: ${error}`);
    } else {
      console.log(`[feishu-to-wechat] 已转发: ${sender} -> ${targetUser}`);
    }
  } catch (error) {
    console.error(`[feishu-to-wechat] 发送异常:`, error);
  }
}

function shouldForward(context: MessageContext): boolean {
  // 关键词过滤
  const keywords = process.env.FILTER_KEYWORDS;
  if (keywords) {
    const keywordList = keywords.split(",").map(k => k.trim().toLowerCase());
    const content = context.content.toLowerCase();
    // 如果配置了关键词，但消息不包含任何关键词，则不转发
    if (keywordList.length > 0 && !keywordList.some(k => content.includes(k))) {
      return false;
    }
  }

  return true;
}

const handler = async (event: { type: string; action: string; context: MessageContext }) => {
  // 只处理消息接收事件
  if (event.type !== "message" || event.action !== "received") {
    return;
  }

  const { context } = event;

  // 只转发飞书消息
  if (context.channelId !== "feishu") {
    return;
  }

  // 检查是否应该转发
  if (!shouldForward(context)) {
    return;
  }

  // 异步转发，不阻塞主流程
  void forwardToWeChat(context);
};

export default handler;
````

```typescript
if (processedIds.has(context.messageId || "")) {
  return; // 已处理
}
processedIds.add(context.messageId || "");
```

### 9.4 失败重试

```typescript
async function forwardWithRetry(context: MessageContext, maxRetries = 3) {
  for (let i = 0; i < maxRetries; i++) {
    try {
      await forwardToWeChat(context);
      return;
    } catch (error) {
      if (i === maxRetries - 1) throw error;
      await new Promise((r) => setTimeout(r, 5000 * (i + 1))); // 指数退避
    }
  }
}
```
